Feature skeleton

@Module({
  controllers: [TicketsController],
  providers: [TicketsService],
  exports: [TicketsService],
})
export class TicketsModule {}

@Controller('tickets')
export class TicketsController {
  constructor(private readonly tickets: TicketsService) {}

  @Patch(':id')
  update(
    @Param('id', ParseIntPipe) id: number,
    @Body() dto: UpdateTicketDto,
  ) { return this.tickets.update(id, dto); }
}

@Injectable()
export class TicketsService {}

Handler decorators

  • @Get @Post @Patch @Put @Delete
  • @Param('id') always a string, parse it
  • @Query() @Body() @Headers('x')
  • @HttpCode(204) @Header(k, v)
  • Skip @Req @Res: they disable interceptors
  • Static routes before :id routes

Dependency injection

providers: [
  { provide: 'FROM', useValue: 'a@b.dev' },
  { provide: Mailer, useClass: SmtpMailer },
  { provide: 'DB', inject: [ConfigService],
    useFactory: (c: ConfigService) => connect(c.get('DB_URL')) },
]
constructor(@Inject('FROM') from: string) {}

Default scope = one shared instance. Request scope is slow; avoid.

Validation

app.useGlobalPipes(new ValidationPipe({
  whitelist: true,
  forbidNonWhitelisted: true,  // unknown field = 400
  transform: true,
}));

class CreateTicketDto {
  @IsString() @IsNotEmpty() subject: string;
  @IsIn(['low','medium','high']) priority: string;
  @IsOptional() @IsString() description?: string;
}

Undecorated fields get stripped. Or use Zod: @Body({ schema }).

Guard + metadata

export const Roles = (...r: string[]) => SetMetadata('roles', r);

canActivate(ctx: ExecutionContext) {
  const roles = this.reflector.getAllAndOverride('roles',
    [ctx.getHandler(), ctx.getClass()]);
  if (!roles) return true;
  return roles.includes(
    ctx.switchToHttp().getRequest().user?.role);
}

Global, with DI

providers: [
  { provide: APP_GUARD, useClass: AuthGuard },
  { provide: APP_PIPE, useClass: ValidationPipe },
  { provide: APP_FILTER, useClass: AllExceptionsFilter },
  { provide: APP_INTERCEPTOR, useClass: TimingInterceptor },
]

Prefer these over useGlobal* in main.ts.

Interceptor / decorator

intercept(ctx: ExecutionContext, next: CallHandler) {
  const t = Date.now();
  return next.handle().pipe(
    tap(() => this.logger.log(`${Date.now() - t}ms`)));
}

export const CurrentUser = createParamDecorator(
  (_: unknown, ctx: ExecutionContext) =>
    ctx.switchToHttp().getRequest().user);

  • db typeorm drizzle mongoose sequelize
  • retries @nestjs/resilience
  • also: mail, storage, i18n, locks, outbox, webhooks, graphql, websockets, microservices

Do

  • One module per feature, thin controllers
  • DTO or schema on every input
  • Throw HTTP exceptions, format in one filter
  • Return response shapes, not raw DB rows
  • Paginate lists, transaction on multi-step writes
  • app.enableShutdownHooks() and a health endpoint

Avoid

  • new Service(): inject instead
  • Business logic in controllers
  • @Global and forwardRef as shortcuts
  • Scattered process.env
  • Adding a library before checking Nest's own packages
